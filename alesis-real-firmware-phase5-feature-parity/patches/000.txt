/* Real SR-16 firmware scans the key matrix more slowly than HR-16/HR-16B.
   The old 18 ms sequencer pulse was routinely missed. */
p2Pulse=function(name,hold=18){
  const mix=p2Mix[name];if(mix?.mute)return;
  const q=p2Coord(name);if(!q)return;
  p2QueuePad(name);
  const[c,b]=q;
  const minimum=(typeof activeMachine!=='undefined'&&activeMachine==='sr16')?85:24;
  const effective=Math.max(minimum,hold||0);
  columns[c]&=~(1<<b);
  setTimeout(()=>{columns[c]|=(1<<b)},effective);
};

  ph4PrimeContext();
  if(typeof activeMachine!=='undefined'&&activeMachine==='sr16'&&cpu&&cpu.cycles<2550000){
    if(ph4PendingPlay)return;
    const wait=Math.max(120,Math.min(2600,Math.ceil((2550000-cpu.cycles)/1000)+140));
    const st=document.getElementById('p2PatternStatus');if(st)st.textContent=`SR-16 REAL FIRMWARE BOOTING • PATTERN STARTS IN ${(wait/1000).toFixed(1)}s`;
    ph4PendingPlay=setTimeout(()=>{ph4PendingPlay=0;p2Play()},wait);
    return;
  }
  const out=ph4OldPlay.apply(this,arguments);
  if(p2Playing&&ph4ResumeStep!=null){p2Step=((ph4ResumeStep%p2Length)+p2Length)%p2Length;ph4ResumeStep=null}
  return out;
};
